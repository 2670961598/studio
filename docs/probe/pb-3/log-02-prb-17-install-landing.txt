# ---- 附：探针脚本源（仓外临时件；非原始输出；运行前设 REPO_ROOT=<repo>） ----
/**
 * PB-3 / PRB-17 HCL-0A 落位面探针（只写系统临时目录）
 * 1) importDefinitionPackage：定义包落位 → 检查返回记录/目标目录是否有「宿主未承认」明记与安装日志
 * 2) installCompanyPlugins：插件落位+登记 → 检查 install-log.jsonl 是否有「宿主未承认」明记
 * 输出不含本机路径。
 */
import { mkdtemp, mkdir, writeFile, readdir, readFile, rm } from 'node:fs/promises'
import { existsSync } from 'node:fs'
import { tmpdir } from 'node:os'
import path from 'node:path'

const REPO = process.env.REPO_ROOT
if (!REPO) throw new Error('REPO_ROOT 未设')
const { importDefinitionPackage } = await import(`${REPO}/packages/company/src/assets/index.ts`)
const { installCompanyPlugins } = await import(`${REPO}/packages/company/src/dependencies.ts`)

const root = await mkdtemp(path.join(tmpdir(), 'sn-pb3-17-'))
const markerRe = /宿主未承认|未加载|未承认|host.*unrecogni|host.*unacknowledg/i
try {
  // ── 1. PKG-DEF 能力包导入 ──
  const src = path.join(root, 'pkg-src')
  const target = path.join(root, 'pkg-target')
  await mkdir(src, { recursive: true })
  await writeFile(path.join(src, 'package.yaml'), [
    'schema: studio-next/pkg-def@1',
    'artifactId: company.pb3-probe',
    'artifactVersion: 1.0.0',
    'specVersion: 2.0.0',
    'parts:',
    '  - {partId: definition.main, partType: definition}',
    '',
  ].join('\n'))
  await writeFile(path.join(src, 'org-config.yaml'), [
    'schema: org-config/v1',
    'company:',
    '  name: PB3 探针公司',
    '  slug: pb3-probe',
    '  level: 2',
    '',
  ].join('\n'))
  await writeFile(path.join(src, '.studio-next.yaml'), [
    'schema: studio-next/org@1',
    'template: company-2-level',
    '',
  ].join('\n'))

  const structureFacts = {
    established: true,
    rootDelegation: true,
    bossCovered: true,
    inPositionLevels: [],
    deptCount: 0,
    maxDeptDepth: 1,
    auditPosition: false,
  }
  const report = await importDefinitionPackage({ sourceDir: src, targetDir: target, structureFacts })
  console.log('== importDefinitionPackage ==')
  console.log(`artifactId=${report.artifactId} version=${report.artifactVersion} specVersion=${report.specVersion}`)
  console.log(`parts=${JSON.stringify(report.parts)} files=${JSON.stringify(report.files)}`)
  console.log(`installAuthorization=${report.installAuthorization} staged=${report.staged}`)
  console.log(`scanNote=${JSON.stringify(report.scanNote)}`)
  const reportText = JSON.stringify(report)
  const markerInReport = markerRe.test(reportText)
  console.log(`report marker(宿主未承认/未加载) = ${markerInReport}`)
  const targetEntries = (await readdir(target)).sort()
  console.log(`target entries = ${JSON.stringify(targetEntries)}`)
  console.log(`target install-log.jsonl exists = ${existsSync(path.join(target, 'install-log.jsonl'))}`)
  console.log(`target registry.json exists = ${existsSync(path.join(target, 'registry.json'))}`)
  const targetTexts = []
  for (const rel of targetEntries.filter((e) => e.endsWith('.yaml') || e.endsWith('.md') || e.endsWith('.json'))) {
    targetTexts.push(`${rel}:${markerRe.test(await readFile(path.join(target, rel), 'utf8'))}`)
  }
  console.log(`target file marker scan = ${JSON.stringify(targetTexts)}`)

  // ── 2. 插件安装日志面 ──
  const companyDir = path.join(root, 'company')
  const pluginRoot = path.join(root, 'plugin-root')
  const pluginName = 'dsh-pb3-probe'
  await mkdir(path.join(companyDir, 'plugins', pluginName, 'src'), { recursive: true })
  await writeFile(path.join(companyDir, 'dependencies.yaml'), [
    'schema: studio-next/dependencies@1',
    'bindings:',
    '  - when: { attr: role, is: dev }',
    '    plugins:',
    `      - { name: ${pluginName}, version: 1.0.0 }`,
    '',
  ].join('\n'))
  await writeFile(path.join(companyDir, 'plugins', pluginName, 'package.json'), `${JSON.stringify({ name: pluginName, version: '1.0.0', type: 'module', dsh: { bundle: { patch: './cordis.patch.yml' } } }, null, 2)}\n`)
  await writeFile(path.join(companyDir, 'plugins', pluginName, 'cordis.patch.yml'), `- insert:\n    - id: ${pluginName}\n      name: ./src/index.mjs\n      config: {}\n`)
  await writeFile(path.join(companyDir, 'plugins', pluginName, 'src', 'index.mjs'), `export const name = '${pluginName}'\n`)
  const persons = [{ id: 'AGENT-1', dept: '工程部', level: 2, role: 'dev' }]
  const install = await installCompanyPlugins({ companyDir, pluginRoot, persons, by: 'probe', now: () => new Date('2026-10-29T00:00:00Z') })
  console.log('== installCompanyPlugins ==')
  console.log(`confirmed=${install.confirmed} items=${JSON.stringify(install.items.map((i) => `${i.name}:${i.action}`))}`)
  console.log(`log exists = ${existsSync(install.logFile)}`)
  const logText = existsSync(install.logFile) ? await readFile(install.logFile, 'utf8') : ''
  console.log(`install-log line count = ${logText.trim() === '' ? 0 : logText.trim().split('\n').length}`)
  console.log(`install-log marker(宿主未承认/未加载) = ${markerRe.test(logText)}`)
  console.log(`install-log first line = ${logText.trim().split('\n')[0] ?? ''}`)
} finally {
  await rm(root, { recursive: true, force: true })
}
