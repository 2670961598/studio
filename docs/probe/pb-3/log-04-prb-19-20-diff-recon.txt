/**
 * PB-3 / PRB-19/20 侦察腿（不构成本行探针证据）
 * 用 PRM 生成器编译新旧两份计划，列出 tools/authority/sandbox 段引用与 secretTypes 的可比对差；
 * 同时打印「无 permission_diff/secret_diff 运行期生成器」「无部署供给确认」的静态事实。
 * 输出不含本机路径。
 */
const REPO = process.env.REPO_ROOT
if (!REPO) throw new Error('REPO_ROOT 未设')
const { compileProjection } = await import(`${REPO}/packages/core/src/prm/compile-projection.ts`)

const allSegments = ['identity', 'authority', 'plan', 'tools', 'sandbox', 'approval', 'evidence', 'persona', 'validity']
const base = {
  companyId: 'company-probe',
  workRef: 'work-1',
  planRef: 'plan-old',
  grantRefs: ['grant-a', 'grant-b'],
  capabilityRefs: ['cap-a'],
  host: { hostKind: 'dsh', version: '0.2.0-rc.2', bindingRef: 'BIND-DSH-01' },
  secretTypes: ['per_task_shortlived'],
  validityFrom: '2026-11-01T00:00:00Z',
  validityUntil: '2026-12-01T00:00:00Z',
  deployment: {
    carriedSegments: allSegments,
    sandbox: { mode: 'workspace-write', rootRef: 'sandbox-root-old' },
    approvalTemplateRef: 'approval-template',
    evidenceSinkRef: 'evidence-sink',
    personaRefs: ['persona-a'],
  },
}
const oldInput = base
const newInput = {
  ...base,
  planRef: 'plan-new',
  grantRefs: ['grant-a', 'grant-c'],
  capabilityRefs: ['cap-a', 'cap-b'],
  secretTypes: ['per_task_shortlived', 'platform_token'],
  deployment: {
    ...base.deployment,
    sandbox: { mode: 'workspace-write', rootRef: 'sandbox-root-new' },
  },
}

function compile(input) {
  const r = compileProjection(input)
  if (r.kind !== 'ok') throw new Error(`compileProjection err ${r.error.subcause}`)
  return r.value
}
const oldP = compile(oldInput)
const newP = compile(newInput)

function pick(p, seg) {
  return { enabled: p.segments[seg].enabled, sourceRefs: p.segments[seg].sourceRefs, note: p.segments[seg].note ?? null }
}
for (const seg of ['plan', 'tools', 'authority', 'sandbox', 'validity']) {
  console.log(`== segment ${seg} ==`)
  console.log(`old: ${JSON.stringify(pick(oldP, seg))}`)
  console.log(`new: ${JSON.stringify(pick(newP, seg))}`)
}
const refs = (p, seg) => new Set(p.segments[seg].sourceRefs)
for (const seg of ['tools', 'authority', 'sandbox', 'plan']) {
  const a = refs(oldP, seg)
  const b = refs(newP, seg)
  const added = [...b].filter((x) => !a.has(x)).sort()
  const removed = [...a].filter((x) => !b.has(x)).sort()
  console.log(`diff ${seg}: added=${JSON.stringify(added)} removed=${JSON.stringify(removed)}`)
}
console.log(`digest old=${oldP.digest}`)
console.log(`digest new=${newP.digest}`)
console.log(`unsupportedSegments old=${JSON.stringify(oldP.unsupportedSegments)} new=${JSON.stringify(newP.unsupportedSegments)}`)
