# ---- 附：探针脚本源（仓外临时件；非原始输出；运行前设 REPO_ROOT=<repo>） ----
/**
 * PB-3 / PRB-18 HCL-1 配置可观察探针（组件级控制面；只写系统临时目录）
 * 复制 2 级公司模板到临时目录，起 createHotReload 控制面组件（不重启进程），
 * 通过 control/reload.request.json 手动触发两次声明变更，观察回执+指纹+台账事件。
 * 输出不含本机路径（log 回调内脱敏临时根）。
 */
import { cp, mkdtemp, readFile, rm, appendFile } from 'node:fs/promises'
import { tmpdir } from 'node:os'
import path from 'node:path'

const REPO = process.env.REPO_ROOT
if (!REPO) throw new Error('REPO_ROOT 未设')
const { createHotReload, writeReloadRequest } = await import(`${REPO}/apps/daemon/src/control/hot-reload.ts`)
const { makeUlid } = await import(`${REPO}/apps/daemon/src/engine/tools/id.ts`)
const { openLedger } = await import(`${REPO}/packages/store/src/index.ts`)
const { loadCompanyDirectory, readDirectoryFiles } = await import(`${REPO}/packages/company/src/index.ts`)

const root = await mkdtemp(path.join(tmpdir(), 'sn-pb3-18-'))
const companyDir = path.join(root, 'company')
const rootDir = path.join(root, 'daemon-root')
await cp(`${REPO}/templates/company-2-level`, companyDir, { recursive: true })
const redact = (line) => line.split(root).join('<tmp>')

const clockStart = Date.parse('2026-10-29T00:00:00Z')
let tick = 0
const ledger = openLedger(path.join(root, 'ledger.db'))
let applied = 0
const applySnapshots = []

try {
  const hot = createHotReload({
    companyDir,
    rootDir,
    ledger,
    company: 'probe-company',
    clock: () => clockStart + (tick += 1),
    makeId: makeUlid,
    log: (line) => console.log(`log> ${redact(line)}`),
    apply: (snapshot) => {
      applied += 1
      applySnapshots.push({ slug: snapshot.slug, level: snapshot.level, people: snapshot.people.length, departments: snapshot.departments.length })
      return { jobsSeeded: 0, jobsKept: 0, jobsRemoved: 0 }
    },
    deptFallback: '工程部',
    watch: false,
    anchorAbsence: { declaredBy: 'pb3-probe' },
  })
  console.log(`baseline fingerprint = ${hot.stats.lastFingerprint}`)

  // 第一次声明变更：charter.md 追加一行
  await appendFile(path.join(companyDir, 'charter.md'), '\n<!-- PRB-18 probe change #1 -->\n')
  writeReloadRequest(rootDir, { id: 'prb18-req-1', company: null, requestedAt: Date.now() })
  const r1 = await hot.check()
  console.log(`check#1 kind=${r1?.kind ?? 'null'} fingerprint=${r1?.fingerprint ?? '-'} changes=${JSON.stringify(r1?.changes ?? '')}`)
  const result1 = JSON.parse(await readFile(path.join(rootDir, 'control', 'reload.result.json'), 'utf8'))
  console.log(`result#1 = ${JSON.stringify(result1)}`)

  // 第二次声明变更：prompts/company.md 追加一行
  await appendFile(path.join(companyDir, 'prompts', 'company.md'), '\n<!-- PRB-18 probe change #2 -->\n')
  writeReloadRequest(rootDir, { id: 'prb18-req-2', company: null, requestedAt: Date.now() })
  const r2 = await hot.check()
  console.log(`check#2 kind=${r2?.kind ?? 'null'} fingerprint=${r2?.fingerprint ?? '-'} changes=${JSON.stringify(r2?.changes ?? '')}`)
  const result2 = JSON.parse(await readFile(path.join(rootDir, 'control', 'reload.result.json'), 'utf8'))
  console.log(`result#2 = ${JSON.stringify(result2)}`)

  // 第三次无变更：控制面仍在，无动作
  const r3 = await hot.check()
  console.log(`check#3 kind=${r3?.kind ?? 'null'}`)

  console.log(`stats checks=${hot.stats.checks} reloads=${hot.stats.reloads} manualRequests=${hot.stats.manualRequests} lastFingerprint=${hot.stats.lastFingerprint}`)
  console.log(`apply calls=${applied} snapshots=${JSON.stringify(applySnapshots)}`)
  console.log(`fp1!=fp2 = ${r1?.fingerprint !== r2?.fingerprint}`)
  const events = ledger.queryEvents({ company: 'probe-company' })
  console.log(`ledger events=${events.length} eventIds=${JSON.stringify(events.map((e) => e.id))}`)
  for (const e of events) console.log(`event ${e.event}: ${redact(String(e.note)).slice(0, 200)}`)

  const mounted = loadCompanyDirectory(await readDirectoryFiles(companyDir))
  console.log(`loadCompanyDirectory after reload: ok=${mounted.ok}${mounted.ok ? '' : ` issues=${mounted.issues.length}`}`)
} finally {
  ledger.close()
  await rm(root, { recursive: true, force: true })
}
