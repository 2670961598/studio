/**
 * PB-5 / PRB-23 映射腿探针（只读；不写仓内文件）
 * 运行：REPO_ROOT=<repo> node pb5-map.mjs
 * 输出只含相对标识/状态值，不打印本机绝对路径。
 */
const REPO = process.env.REPO_ROOT
if (!REPO) throw new Error('REPO_ROOT 未设')

const yt = await import(`${REPO}/packages/yt/src/binding.ts`)
const taiga = await import(`${REPO}/packages/taiga/src/binding.ts`)
const local = await import(`${REPO}/packages/localplatform/src/binding.ts`)

const platforms = [
  ['yt', yt],
  ['taiga', taiga],
  ['localplatform', local],
]

const cases = [
  ['legal-unmapped-ascii', 'In Progress'],
  ['legal-unmapped-workstate-literal', 'in_progress'],
  ['legal-unmapped-accepted-literal', 'accepted'],
  ['legal-unmapped-raw-whitespace', '  In Progress  '],
  ['empty', ''],
  ['whitespace-only', '   '],
  ['non-string-null', null],
  ['non-string-number', 42],
]

for (const [name, mod] of platforms) {
  const profile = mod.bindingProfile()
  const p = profile.kind === 'ok' ? profile.value : null
  console.log(`== ${name} ==`)
  console.log(`bindingProfile: kind=${profile.kind} bindingId=${p?.bindingId ?? '-'} stateMappings=${p?.stateMappings.length ?? '-'} probes=${JSON.stringify(p?.probes?.map((x) => `${x.probeId}:${x.state}`))}`)
  for (const [label, value] of cases) {
    let out
    try {
      out = mod.mapState(value)
    } catch (err) {
      out = { kind: 'throw', error: String(err) }
    }
    const r = out.kind === 'ok'
      ? { kind: 'ok', platformState: out.value.platformState, workStates: out.value.workStates, verification: out.value.verification, unmappedBehavior: out.value.unmappedBehavior }
      : { kind: 'err', category: out.error?.category, subcause: out.error?.subcause }
    console.log(`${label}: ${JSON.stringify(r)}`)
  }
}
